### Task 5: Export Functionality (30 minutes)
**Goal:** Allow users to download work products

```typescript
// In SessionDetails.tsx, add export buttons:

const ExportButtons = ({ sessionId }: { sessionId: string }) => {
  const handleExport = async (format: 'txt' | 'json') => {
    try {
      const response = await fetch(`/api/simple/recording-sessions/${sessionId}/download/transcript`);
      const blob = await response.blob();
      
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `meeting-${sessionId}.${format}`;
      a.click();
      URL.revokeObjectURL(url);
    } catch (error) {
      console.error('Export failed:', error);
    }
  };

  return (
    <div className="flex gap-2">
      <button 
        onClick={() => handleExport('txt')}
        className="bg-blue-600 text-white px-3 py-1 rounded text-sm"
      >
        Export Transcript
      </button>
      <button 
        onClick={() => handleExport('json')}
        className="bg-green-600 text-white px-3 py-1 rounded text-sm"
      >
        Export Summary
      </button>
    </div>
  );
};
```
